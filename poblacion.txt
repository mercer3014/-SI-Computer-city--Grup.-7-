DELETE FROM valor_atributo
WHERE valor = 'Bluetooth' AND atributo_id = (SELECT id FROM atributo WHERE nombre='Conectividad');
 
-- 4.2 Baja lógica de un producto discontinuado (preferida sobre DELETE físico,
--     ya que producto está referenciado por variante_producto/ventas/compras)
UPDATE producto SET estado = 'INACTIVO' WHERE slug = 'barra-de-sonido-jbl-bar-2-0';
 
-- 4.3 Eliminación física de un registro que sí es seguro borrar por completo:
--     una cotización de prueba que nunca fue enviada al cliente
-- (Se muestra el patrón completo: primero el detalle, luego el encabezado,
--  respetando las llaves foráneas; ON DELETE CASCADE también lo permitiría directo)
DELETE FROM cotizacion WHERE codigo_cotizacion = 'COT-PRUEBA-000' ;
 
-- 4.4 Anulación de una venta (NO se hace con DELETE, se usa el procedimiento
--     sp_anular_venta para no perder trazabilidad contable ni de garantías)
DO $$
DECLARE
    v_usuario_id INT;
BEGIN
    SELECT id INTO v_usuario_id FROM usuario WHERE email = 'ruben.morales@computercity.com';
 
    CALL sp_anular_venta(
        p_numero_venta => 'V-2026-0002',
        p_usuario_id   => v_usuario_id,
        p_motivo       => 'Venta registrada por error, cliente no llegó a pagar'
    );
END;
$$;
